def one_close(s,n=1,i=None):
    if i is None: i=1
    l = len(s)
    repeaters = set()
    for i in range(1, l):
        for j in (i, l-1):
            if i <= n:
                repeaters.update(set(map("".join, zip(s[:n+i],s[i:n+j+1]))))
            elif i > n:
                repeaters.update(set(map("".join, zip(s[:j-i+1],s[j-n:j+1]))))
            else:
                break
    return list(repeaters)
